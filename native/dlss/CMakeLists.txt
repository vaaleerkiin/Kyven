cmake_minimum_required(VERSION 3.21)
project(KyvenDLSSWorker VERSION 1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Kyven DLSS uses Windows D3D12")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(KyvenDLSSWorker main.cpp KyvenDlssEngine.cpp)
add_library(nvngx SHARED caller_shim.cpp)
foreach(target KyvenDLSSWorker nvngx)
  target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_link_libraries(${target} PRIVATE d3d12 dxgi dxguid)
endforeach()
install(TARGETS KyvenDLSSWorker nvngx RUNTIME DESTINATION .)
install(FILES LICENSE.upstream DESTINATION .)
